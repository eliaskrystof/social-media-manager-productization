/*******************************
 * UPDATE ACCESS TOKENS
 *******************************/
function updateAccessTokens() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(TOKENS_SHEET);
  if (!sheet) {
    logToSheet("error", `Sheet '${TOKENS_SHEET}' not found.`, "access_tokens", "", true);
    return;
  }

  const docId = ss.getId();
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  const values = sheet.getRange(2, 1, 1, sheet.getLastColumn()).getValues()[0];

  const COLUMN_MAP = {
    instagram_access_token: "ig_access_token",
    facebook_access_token: "fb_access_token",
    linkedin_access_token: "linkedin_user_token",
    instagram_id: "ig_user_id",
    facebook_id: "fb_page_id",
    linkedin_id: "linkedin_user_id"
  };

  const payload = {};
  headers.forEach((header, i) => {
    const supabaseField = COLUMN_MAP[header];
    if (supabaseField) payload[supabaseField] = values[i];
  });

  if (Object.keys(payload).length === 0) {
    logToSheet("info", "No mapped fields found to update in Supabase.", "access_tokens", "", true);
    return;
  }

  const url = `${SUPABASE_URL}/rest/v1/${SUPABASE_TABLE}?master_table_id=eq.${docId}`;
  try {
    const response = UrlFetchApp.fetch(url, {
      method: "PATCH",
      headers: {
        "apikey": SUPABASE_KEY,
        "Authorization": "Bearer " + SUPABASE_KEY,
        "Content-Type": "application/json",
        "Prefer": "return=representation"
      },
      payload: JSON.stringify(payload)
    });

    const result = JSON.parse(response.getContentText());
    Logger.log("Supabase response: " + JSON.stringify(result, null, 2));
    logToSheet("success", "Supabase updated successfully!", "access_tokens", "", false);
  } catch (err) {
    logToSheet("error", "Supabase update failed: " + err.message, "access_tokens", "", true);
  }
}

/*******************************
 * VALIDATE ACCESS TOKENS
 *******************************/
function validateTokens() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(TOKENS_SHEET);
  if (!sheet) {
    logToSheet("error", `Sheet '${TOKENS_SHEET}' not found.`, "access_tokens", "", true);
    return;
  }

  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  const values = sheet.getRange(2, 1, 1, sheet.getLastColumn()).getValues()[0];
  const data = {};
  headers.forEach((h, i) => data[h] = values[i]);

  const instagramToken = data["instagram_access_token"];
  const facebookToken = data["facebook_access_token"];
  const linkedinToken = data["linkedin_access_token"];

  const tokenResults = {
    instagram: checkInstagramToken(instagramToken),
    facebook: checkFacebookToken(facebookToken),
    linkedin: checkLinkedInToken(linkedinToken)
  };

  const tokenCols = {
    instagram: headers.indexOf("instagram_access_token") + 1,
    facebook: headers.indexOf("facebook_access_token") + 1,
    linkedin: headers.indexOf("linkedin_access_token") + 1,
  };

  const range = sheet.getRange(2, tokenCols.instagram, 1, 3);
  const colors = [
    tokenResults.instagram ? "#b7e1cd" : "#f4c7c3",
    tokenResults.facebook ? "#b7e1cd" : "#f4c7c3",
    tokenResults.linkedin ? "#b7e1cd" : "#f4c7c3"
  ];
  range.setBackgrounds([colors]);

  logToSheet("info", "Validation complete!", "access_tokens", "", false);
}

/*******************************
 * TOKEN VALIDATORS
 *******************************/
function checkFacebookToken(token) {
  if (!token) return false;
  try {
    const url = `https://graph.facebook.com/me?fields=id,name&access_token=${encodeURIComponent(token)}`;
    const res = UrlFetchApp.fetch(url);
    const json = JSON.parse(res.getContentText());
    return json.id ? true : false;
  } catch (e) {
    logToSheet("error", "Facebook token check failed: " + e.message, "access_tokens", "", false);
    return false;
  }
}

function checkInstagramToken(token) {
  if (!token) return false;
  try {
    const url = `https://graph.facebook.com/v18.0/me?fields=id,name&access_token=${encodeURIComponent(token)}`;
    const res = UrlFetchApp.fetch(url, { muteHttpExceptions: true });
    const text = res.getContentText();
    if (res.getResponseCode() !== 200) {
      logToSheet("error", `Instagram Graph API returned ${res.getResponseCode()}: ${text}`, "access_tokens", "", false);
      return false;
    }
    const json = JSON.parse(text);
    return json.id ? true : false;
  } catch (e) {
    logToSheet("error", "Instagram token check failed: " + e.message, "access_tokens", "", false);
    return false;
  }
}

function checkLinkedInToken(token) {
  if (!token) return false;
  try {
    const url = "https://api.linkedin.com/v2/userinfo";
    const options = {
      method: "get",
      headers: {
        "Authorization": "Bearer " + token,
        "Accept": "application/json",
        "X-Restli-Protocol-Version": "2.0.0",
        "LinkedIn-Version": "202401"
      },
      muteHttpExceptions: true
    };
    const res = UrlFetchApp.fetch(url, options);
    const code = res.getResponseCode();
    if (code !== 200) {
      logToSheet("error", `LinkedIn returned ${code}: ${res.getContentText()}`, "access_tokens", "", false);
      return false;
    }
    const json = JSON.parse(res.getContentText());
    return !!(json.sub || json.id);
  } catch (e) {
    logToSheet("error", "LinkedIn token check failed: " + e.message, "access_tokens", "", false);
    return false;
  }
}
