/**********************
 * SERVER-SIDE: PROCESS SELECTED LANGUAGES (CREATE/SEND)
 **********************/

// Called from the HTML dialog with an array like ['czech', 'german'] or ['all']
function processNewsletterLanguages(selectedLangs) {
  var ui = SpreadsheetApp.getUi();

  // Resolve language codes
  var langCodes = [];

  if (selectedLangs.indexOf('all') !== -1) {
    langCodes = Object.values(LANGUAGE_CODES); // all languages
  } else {
    selectedLangs.forEach(function(name) {
      var code = LANGUAGE_CODES[name];
      if (code) {
        langCodes.push(code);
      }
    });
  }

  if (langCodes.length === 0) {
    ui.alert('No valid language selected.');
    return;
  }

  var ss = SpreadsheetApp.getActive();
  var sheet = ss.getSheetByName(NL_SHEET_NAME);

  if (!sheet) {
    ui.alert('Sheet "' + NL_SHEET_NAME + '" not found.');
    return;
  }

  var lastRow = sheet.getLastRow();
  var lastCol = sheet.getLastColumn();

  if (lastRow <= HEADER_ROW) {
    ui.alert('No data rows found.');
    return;
  }

  var dataRange = sheet.getRange(HEADER_ROW + 1, 1, lastRow - HEADER_ROW, lastCol);
  var values = dataRange.getValues();

  var settings = loadSettings_();
  var payloadArray = [];

  // Group by row, include rowNumber at main level
  values.forEach(function(row, index) {
    var isChecked = row[CHECKBOX_COLUMN - 1] === true;
    if (!isChecked) return;

    // Actual 1-based sheet row number for this data row
    var sheetRowNumber = HEADER_ROW + 1 + index;

    var rowEntry = {
      rowNumber: sheetRowNumber,
      languages: []
    };

    langCodes.forEach(function(langCode) {
      var obj = buildJsonFromRow_(row, settings, langCode);
      obj.language = langCode;
      rowEntry.languages.push(obj);
    });

    payloadArray.push(rowEntry);
  });

  if (payloadArray.length === 0) {
    ui.alert('No checked rows found for the selected language(s).');
    return;
  }

  var payload = {
    newsletters: payloadArray
  };

  try {
    var response = UrlFetchApp.fetch(WEBHOOK_URL, {
      method: 'post',
      contentType: 'application/json',
      payload: JSON.stringify(payload),
      muteHttpExceptions: true
    });

    ui.alert(
      'POST sent.\nHTTP Status: ' + response.getResponseCode() +
      '\nResponse: ' + response.getContentText()
    );

  } catch (e) {
    ui.alert('Error sending POST: ' + e);
  }
}

// Called from Edit language dialog
function processNewsletterEditLanguages(selectedLangs) {
  var ui = SpreadsheetApp.getUi();

  // Resolve language codes
  var langCodes = [];

  if (selectedLangs.indexOf('all') !== -1) {
    langCodes = Object.values(LANGUAGE_CODES); // all languages
  } else {
    selectedLangs.forEach(function(name) {
      var code = LANGUAGE_CODES[name];
      if (code) {
        langCodes.push(code);
      }
    });
  }

  if (langCodes.length === 0) {
    ui.alert('No valid language selected.');
    return;
  }

  var ss = SpreadsheetApp.getActive();
  var sheet = ss.getSheetByName(NL_SHEET_NAME);

  if (!sheet) {
    ui.alert('Sheet "' + NL_SHEET_NAME + '" not found.');
    return;
  }

  var lastRow = sheet.getLastRow();
  var lastCol = sheet.getLastColumn();

  if (lastRow <= HEADER_ROW) {
    ui.alert('No data rows found.');
    return;
  }

  var dataRange = sheet.getRange(HEADER_ROW + 1, 1, lastRow - HEADER_ROW, lastCol);
  var values = dataRange.getValues();

  // Find checked rows
  var checkedRows = [];
  values.forEach(function(row, index) {
    if (row[CHECKBOX_COLUMN - 1] === true) {
      checkedRows.push(HEADER_ROW + 1 + index); // sheet row number
    }
  });

  if (checkedRows.length === 0) {
    ui.alert('No checked row found. Please check exactly one row to edit.');
    return;
  }

  if (checkedRows.length > 1) {
    ui.alert('Please check only one row for edit.');
    return;
  }

  var sheetRowNumber = checkedRows[0];

  // Show second dialog to select sections + comments, per language (wizard)
  showEditSectionsDialog_(sheetRowNumber, langCodes);
}

/**********************
 * SERVER-SIDE: SIMPLE ROW + LANGUAGE ACTIONS
 **********************/

function processSimpleActionLanguages(action, selectedLangs) {
  var ui = SpreadsheetApp.getUi();

  // resolve languages
  var langCodes = [];
  if (selectedLangs.indexOf('all') !== -1) {
    langCodes = Object.values(LANGUAGE_CODES);
  } else {
    selectedLangs.forEach(function(name) {
      var code = LANGUAGE_CODES[name];
      if (code) langCodes.push(code);
    });
  }

  if (langCodes.length === 0) {
    return 'No valid language selected.';
  }

  // pick webhook by action
  var url = '';
  if (action === 'odoo') url = ODOO_WEBHOOK_URL;
  else if (action === 'confirm_send') url = CONFIRM_SEND_WEBHOOK_URL;
  else return 'Unknown action: ' + action;

  // sheet + data
  var ss = SpreadsheetApp.getActive();
  var sheet = ss.getSheetByName(NL_SHEET_NAME);
  if (!sheet) return 'Sheet "' + NL_SHEET_NAME + '" not found.';

  var lastRow = sheet.getLastRow();
  var lastCol = sheet.getLastColumn();
  if (lastRow <= HEADER_ROW) return 'No data rows found.';

  var values = sheet.getRange(HEADER_ROW + 1, 1, lastRow - HEADER_ROW, lastCol).getValues();

  // collect checked rows + target group
  var checkedRows = [];
  values.forEach(function(row, index) {
    if (row[COL.CHECKBOX - 1] === true) {
      checkedRows.push({
        rowNumber: HEADER_ROW + 1 + index,
        targetGroup: row[COL.TARGET_GROUP - 1],
        month: row[COL.MONTH - 1]
      });
    }
  });

  if (checkedRows.length === 0) {
    return 'No checked rows found.';
  }

  // send individual POSTs
  var sent = 0;
  var failures = [];

  checkedRows.forEach(function(item) {
    langCodes.forEach(function(langCode) {
      var payload = {
        row_number: item.rowNumber,
        language: langCode,
        target_group: item.targetGroup,
        month: item.month
      };

      try {
        var resp = UrlFetchApp.fetch(url, {
          method: 'post',
          contentType: 'application/json',
          payload: JSON.stringify(payload),
          muteHttpExceptions: true
        });

        var code = resp.getResponseCode();
        if (code >= 200 && code < 300) {
          sent++;
        } else {
          failures.push(
            'Row ' + item.rowNumber + ', ' + langCode + ' → HTTP ' + code + ' → ' + resp.getContentText()
          );
        }
      } catch (e) {
        failures.push(
          'Row ' + item.rowNumber + ', ' + langCode + ' → ERROR: ' + e
        );
      }
    });
  });

  var msg = 'Done. Sent: ' + sent + ' request(s).';
  if (failures.length) {
    msg += '\n\nFailures:\n' + failures.slice(0, 10).join('\n');
    if (failures.length > 10) msg += '\n...and ' + (failures.length - 10) + ' more';
  }
  return msg;
}

/**********************
 * SERVER-SIDE: PROCESS SELECTED LANGUAGES (STRICT EDIT)
 **********************/

function processStrictEditLanguages(selectedLangs) {
  var ui = SpreadsheetApp.getUi();

  // Resolve language codes
  var langCodes = [];

  if (selectedLangs.indexOf('all') !== -1) {
    langCodes = Object.values(LANGUAGE_CODES);
  } else {
    selectedLangs.forEach(function(name) {
      var code = LANGUAGE_CODES[name];
      if (code) {
        langCodes.push(code);
      }
    });
  }

  if (langCodes.length === 0) {
    return 'No valid language selected.';
  }

  var ss = SpreadsheetApp.getActive();
  var sheet = ss.getSheetByName(NL_SHEET_NAME);

  if (!sheet) {
    return 'Sheet "' + NL_SHEET_NAME + '" not found.';
  }

  var lastRow = sheet.getLastRow();
  var lastCol = sheet.getLastColumn();

  if (lastRow <= HEADER_ROW) {
    return 'No data rows found.';
  }

  var values = sheet.getRange(HEADER_ROW + 1, 1, lastRow - HEADER_ROW, lastCol).getValues();

  // Find checked rows
  var checkedRows = [];
  values.forEach(function(row, index) {
    if (row[CHECKBOX_COLUMN - 1] === true) {
      checkedRows.push(HEADER_ROW + 1 + index);
    }
  });

  if (checkedRows.length === 0) {
    return 'No checked row found. Please check exactly one row.';
  }

  if (checkedRows.length > 1) {
    return 'Please check only one row for strict edit.';
  }

  var rowNumber = checkedRows[0];

  var headerMap = getHeaderMap_(sheet);
  
  var sourceRowValue = rowNumber;

  var sent = 0;
  var failures = [];

  langCodes.forEach(function(langCode) {
    var contentHeader = getLangContentHeader_(langCode);
    var contentCol = headerMap[contentHeader];

    if (!contentCol) {
      failures.push('Column "' + contentHeader + '" not found.');
      return;
    }

    var contentBlob = sheet.getRange(rowNumber, contentCol).getValue();

    var payload = {
      source_row: sourceRowValue,
      language: langCode,
      target_languages: [],
      content_blob: contentBlob,
      propagate: false
    };

    try {
      var response = UrlFetchApp.fetch(STRICT_EDIT_WEBHOOK_URL, {
        method: 'post',
        contentType: 'application/json',
        payload: JSON.stringify(payload),
        muteHttpExceptions: true
      });

      var code = response.getResponseCode();
      if (code >= 200 && code < 300) {
        sent++;
      } else {
        failures.push(
          'Language ' + langCode + ' → HTTP ' + code + ' → ' + response.getContentText()
        );
      }

    } catch (e) {
      failures.push('Language ' + langCode + ' → ERROR: ' + e);
    }
  });

  var msg = 'Done. Sent: ' + sent + ' strict edit request(s).';
  if (failures.length) {
    msg += '\n\nFailures:\n' + failures.slice(0, 10).join('\n');
    if (failures.length > 10) msg += '\n...and ' + (failures.length - 10) + ' more';
  }

  return msg;
}

/**********************
 * SUBMIT EDIT REQUEST (SERVER-SIDE)
 **********************/

function submitNewsletterEdit(form) {
  var ui = SpreadsheetApp.getUi();

  try {
    var ss = SpreadsheetApp.getActive();
    var sheet = ss.getSheetByName(NL_SHEET_NAME);

    if (!sheet) {
      throw new Error('Sheet "' + NL_SHEET_NAME + '" not found.');
    }

    var lastCol = sheet.getLastColumn();
    var rowValues = sheet.getRange(form.rowNumber, 1, 1, lastCol).getValues()[0];

    var settings = loadSettings_();

    var rowEntry = {
      rowNumber: form.rowNumber,
      languages: []
    };

    (form.languages || []).forEach(function(langCfg) {
      var langCode = langCfg.langCode;
      var langObj = buildEditJsonFromRow_(rowValues, settings, langCode, langCfg);
      rowEntry.languages.push(langObj);
    });

    var payload = {
      newsletters: [rowEntry]
    };

    var response = UrlFetchApp.fetch(EDIT_WEBHOOK_URL, {
      method: 'post',
      contentType: 'application/json',
      payload: JSON.stringify(payload),
      muteHttpExceptions: true
    });

    var msg =
      'Edit POST sent.\nHTTP Status: ' + response.getResponseCode() +
      '\nResponse: ' + response.getContentText();

    return msg;

  } catch (e) {
    ui.alert('Error sending EDIT POST: ' + e);
    return 'Error sending EDIT POST: ' + e;
  }
}


/**********************
 * SETTINGS LOADER
 **********************/

// Load settings from SETTINGS_SHEET:
// Row 1 = header: Resource | Description | CZ | EN | PL | DE
function loadSettings_() {
  var ss = SpreadsheetApp.getActive();
  var sheet = ss.getSheetByName(SETTINGS_SHEET_NAME);

  if (!sheet) {
    throw new Error('Settings sheet "' + SETTINGS_SHEET_NAME + '" not found.');
  }

  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return {};

  // From row 2 downwards, columns A:F
  var range = sheet.getRange(2, 1, lastRow - 1, 6);
  var values = range.getValues();

  var settings = {};
  values.forEach(function(row) {
    var resource = row[0]; // Col A
    if (!resource) return;

    settings[String(resource).trim()] = {
      cz: row[2], // Col C
      en: row[3], // Col D
      pl: row[4], // Col E
      de: row[5]  // Col F
    };
  });

  return settings;
}

// Get setting by resource key + language code
function getSettingByResourceAndLang_(settings, resourceKey, langCode) {
  if (!resourceKey) return '';

  var trimmedKey = String(resourceKey).trim();
  var entry = settings[trimmedKey];
  if (!entry) return '';

  var prop = LANG_PROP[langCode] || 'cz'; // default to CZ if unknown
  var value = entry[prop];

  return value != null ? value : '';
}

/**********************
 * HEADER MAPPING
 **********************/

function getHeaderMap_(sheet) {
  var lastCol = sheet.getLastColumn();
  var headers = sheet.getRange(HEADER_ROW, 1, 1, lastCol).getValues()[0];

  var map = {};
  headers.forEach(function(header, index) {
    if (!header) return;
    map[String(header).trim()] = index + 1; // 1-based
  });

  return map;
}

function getLangContentHeader_(langCode) {
  var map = {
    czech: 'CZ content',
    english: 'EN content',
    german: 'DE content',
    polish: 'PL content'
  };

  return map[langCode] || '';
}

/**********************
 * JSON BUILDER (UNIFIED KEYS – MAIN SEND)
 **********************/

function buildJsonFromRow_(rowValues, settings, langCode) {
  var obj = {};

  // 1) Direct columns from the row
  obj.month = rowValues[COL.MONTH - 1];
  obj.targetGroup = rowValues[COL.TARGET_GROUP - 1];
  obj.subject = rowValues[COL.SUBJECT - 1];
  obj.blog1 = rowValues[COL.BLOG1 - 1];
  obj.project2 = rowValues[COL.PROJECT2 - 1];

  // 2) Collection (color) block – unified keys
  var colorName = (rowValues[COL.COLOR_COLLECTION - 1] || '').toString().trim();
  var collection = {
    name: colorName || '',
    title: '',
    body: '',
    img1: '',
    img2: '',
    cta: '',
    url: ''
  };

  if (colorName) {
    COLOR_SUFFIXES.forEach(function(suffix) {
      var field = COLOR_SUFFIX_FIELD_MAP[suffix];
      if (!field) return;

      var resourceKey = colorName + suffix;
      var value = getSettingByResourceAndLang_(settings, resourceKey, langCode);

      // image/url keys should always exist (already initialized as empty string)
      if (field === 'img1' || field === 'img2' || field === 'url') {
        collection[field] = value != null ? String(value) : '';
      } else {
        if (value != null && String(value) !== '') {
          collection[field] = String(value);
        }
      }
    });
  }

  obj.collection = collection;

  // 3) Pattern block – unified keys
  var patternName = (rowValues[COL.PATTERN_COLLECTION - 1] || '').toString().trim();
  var pattern = {
    name: patternName || '',
    title: '',
    collectionName: '',
    body: '',
    img1: '',
    img2: '',
    cta: '',
    collectionUrl: ''
  };

  if (patternName) {
    PATTERN_SUFFIXES.forEach(function(suffix) {
      var field = PATTERN_SUFFIX_FIELD_MAP[suffix];
      if (!field) return;

      var resourceKey = patternName + suffix;
      var value = getSettingByResourceAndLang_(settings, resourceKey, langCode);

      if (field === 'img1' || field === 'img2' || field === 'collectionUrl') {
        pattern[field] = value != null ? String(value) : '';
      } else {
        if (value != null && String(value) !== '') {
          pattern[field] = String(value);
        }
      }
    });
  }

  obj.pattern = pattern;

  // 4) CTA content (text) and CTA button (your "link" column)
  var targetGroup = (rowValues[COL.TARGET_GROUP - 1] || '').toString().trim();
  var ctaContentCell = rowValues[COL.CTA_CONTENT - 1];
  var ctaLinkCell = rowValues[COL.CTA_LINK - 1];

  obj.ctaContent = resolveCtaValue_(
    'text',
    targetGroup,
    ctaContentCell,
    settings,
    langCode
  );

  obj.ctaButton = resolveCtaValue_(
    'button',
    targetGroup,
    ctaLinkCell,
    settings,
    langCode
  );

  return obj;
}


/**********************
 * JSON BUILDER (EDIT PAYLOAD – PER LANGUAGE)
 **********************/

// Builds the per-language object for the EDIT webhook.
// Each section has a boolean "change" flag.
// - If change === false → all fields in that section are null.
// - If change === true:
//    - blog/project: include comment + original content
//    - collection/pattern: rebuilt from defaults/settings via buildJsonFromRow_
function buildEditJsonFromRow_(rowValues, settings, langCode, langCfg) {
  var base = buildJsonFromRow_(rowValues, settings, langCode);
  var obj = {};

  obj.language = langCode;

  // Blog section
  var blogChange = !!langCfg.blogChange;
  obj.blog = {
    change: blogChange,
    comment: blogChange ? (langCfg.blogComment || '') : null,
    subject: blogChange ? (base.subject || null) : null,
    blog1: blogChange ? (base.blog1 || null) : null
  };

  // Project section
  var projectChange = !!langCfg.projectChange;
  obj.project = {
    change: projectChange,
    comment: projectChange ? (langCfg.projectComment || '') : null,
    project2: projectChange ? (base.project2 || null) : null
  };

  // Collection block
  var collectionChange = !!langCfg.collectionChange;
  obj.collection = {
    change: collectionChange,
    name: collectionChange ? (base.collection && base.collection.name) || null : null,
    title: collectionChange ? (base.collection && base.collection.title) || null : null,
    body: collectionChange ? (base.collection && base.collection.body) || null : null,
    img1: collectionChange ? (base.collection && base.collection.img1) || null : null,
    img2: collectionChange ? (base.collection && base.collection.img2) || null : null,
    cta: collectionChange ? (base.collection && base.collection.cta) || null : null,
    url: collectionChange ? (base.collection && base.collection.url) || null : null
  };

  // Pattern block
  var patternChange = !!langCfg.patternChange;
  obj.pattern = {
    change: patternChange,
    name: patternChange ? (base.pattern && base.pattern.name) || null : null,
    title: patternChange ? (base.pattern && base.pattern.title) || null : null,
    collectionName: patternChange ? (base.pattern && base.pattern.collectionName) || null : null,
    body: patternChange ? (base.pattern && base.pattern.body) || null : null,
    img1: patternChange ? (base.pattern && base.pattern.img1) || null : null,
    img2: patternChange ? (base.pattern && base.pattern.img2) || null : null,
    cta: patternChange ? (base.pattern && base.pattern.cta) || null : null,
    collectionUrl: patternChange ? (base.pattern && base.pattern.collectionUrl) || null : null
  };

  return obj;
}


/**********************
 * CTA RESOLUTION
 **********************/

// Helper to handle CTA "default" vs custom text
// kind: "text" | "button"
function resolveCtaValue_(kind, targetGroup, cellValue, settings, langCode) {
  var raw = (cellValue != null ? String(cellValue) : '').trim();
  var isDefault =
    raw === '' ||
    raw.toLowerCase() === 'default';

  if (!isDefault) {
    // Use custom cell text directly
    return raw;
  }

  if (!targetGroup) {
    return '';
  }

  var groupConfig = CTA_RESOURCE_KEYS[targetGroup];
  if (!groupConfig) {
    return '';
  }

  var resourceKey =
    kind === 'text' ? groupConfig.textKey : groupConfig.buttonKey;

  return getSettingByResourceAndLang_(settings, resourceKey, langCode);
}
