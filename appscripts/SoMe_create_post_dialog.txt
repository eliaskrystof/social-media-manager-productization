    <script>
      const preparedRows = <?= JSON.stringify(preparedRows) ?>;

      function toggleImageMode() {
        const imageRadios = document.getElementsByName('imageSource');
        let imageSource = "";

        for (let r of imageRadios) {
          if (r.checked) imageSource = r.value;
        }

        const showCreate = imageSource === "create";
        const showReady = imageSource === "yes";

        document.getElementById('create-input').style.display = showCreate ? 'block' : 'none';
        document.getElementById('ready-upload-box').style.display = showReady ? 'block' : 'none';

        if (google.script.host) {
          if (showCreate || showReady) {
            google.script.host.setHeight(620);
          } else {
            google.script.host.setHeight(520);
          }
        }
      }

      function submitSelection() {
        // platform
        const platformRadios = document.getElementsByName('platform');
        let choice = null;
        for (let r of platformRadios) {
          if (r.checked) choice = r.value;
        }

        if (!choice) {
          google.script.run.logToSheet("error", "Platform not selected in createPostDialog.", "create_dialog", "", true);
          return;
        }

        // image source
        const imageRadios = document.getElementsByName('imageSource');
        let image_source = "";
        for (let r of imageRadios) {
          if (r.checked) image_source = r.value;
        }

        if (!image_source) {
          google.script.run.logToSheet("error", "Image source not selected in createPostDialog.", "create_dialog", "", true);
          return;
        }

        const imageInputEl = document.getElementById('imageInput');
        const imageInput = imageInputEl ? imageInputEl.value.trim() : "";

        if (image_source === "create" && !imageInput) {
          google.script.run.logToSheet("error", "Missing image creation description in createPostDialog.", "create_dialog", "", true);
          return;
        }

        google.script.run
          .withSuccessHandler(() => google.script.host.close())
          .withFailureHandler(err => alert(err.message || err))
          .continueCreatePosts({
            choice,
            userInput: image_source === "create" ? imageInput : "",
            image_source,
            preparedRows
          });
      }
    </script>
