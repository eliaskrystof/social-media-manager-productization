create table public.newsletter_campaign_detail (
  id bigint generated by default as identity not null,
  created_at timestamp with time zone not null default now(),
  source_row bigint null,
  campaign_exist boolean null default false,
  campaign_odoo_id bigint null,
  campaign_name character varying null,
  constraint newsletter_campaign_detail_pkey primary key (id),
  constraint newsletter_campaign_detail_source_row_key unique (source_row)
) TABLESPACE pg_default;
