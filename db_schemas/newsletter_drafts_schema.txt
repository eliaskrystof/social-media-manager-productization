create table public.newsletter_drafts (
  post_id bigint generated by default as identity not null,
  created_at timestamp with time zone not null default now(),
  blog_title text null,
  blog_text text null,
  blog_cta_label text null,
  "heroImgBlog" text null,
  project_title text null,
  project_text text null,
  project_cta_label text null,
  "heroImgProject" text null,
  "firstBlock_img" text null,
  "secondBlock_img" text null,
  "sideImgBlog_1" text null,
  "sideImgBlog_2" text null,
  "galleryImgBlog" text null,
  campaign_odoo_id bigint null,
  language text null,
  source_row bigint null,
  collection_title text null,
  collection_text text null,
  collection_cta text null,
  collection_url text null,
  pattern_title text null,
  pattern_text text null,
  pattern_cta text null,
  "patternCollection_name" text null,
  "patternCollection_url" text null,
  cta_text text null,
  cta_button text null,
  final_html text null,
  odoo_post_id bigint null,
  revision bigint null default '0'::bigint,
  draft_key text GENERATED ALWAYS as (
    (
      (
        ((source_row || ':'::text) || language) || ':'::text
      ) || revision
    )
  ) STORED null,
  status text null,
  month text null,
  email_subject text null,
  target_group text null,
  email_from text null,
  blog_url text null,
  project_url text null,
  constraint newsletter_drafts_pkey primary key (post_id)
) TABLESPACE pg_default;

create unique INDEX IF not exists uniq_draft_key on public.newsletter_drafts using btree (draft_key) TABLESPACE pg_default;
